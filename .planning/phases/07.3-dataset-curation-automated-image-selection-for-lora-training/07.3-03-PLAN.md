```typescript
export function useCleanupEvents(operationId: string | null, onDone: (results: ...) => void) {
  useEffect(() => {
    if (!operationId) return
    const es = new EventSource(`/api/v1/cleanup/${operationId}/events`)
    es.addEventListener('cleanup_progress', (e) => { ... })
    es.addEventListener('cleanup_done', (e) => { ... })
    es.addEventListener('cleanup_error', (e) => { ... })
    return () => es.close()
  }, [operationId])
}
```
