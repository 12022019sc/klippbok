```python
"""Model profile schema and bucket generation for target training models.

A model profile defines the resolution, bucketing, caption style, and
training hyperparameter defaults for a target diffusion model. Profiles
are frozen Pydantic v2 models -- overrides produce new instances, they
never mutate the original.

Bucket generation follows the kohya/sd-scripts pixel-budget algorithm:
enumerate all (width, height) pairs within base_resolution^2 pixel budget,
aligned to step_size grid, within dimension bounds.
"""

from __future__ import annotations

import math
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, field_validator, model_validator


class BucketConfig(BaseModel):
    """Parameters for resolution bucket generation."""
    model_config = ConfigDict(frozen=True)

    step_size: int = Field(default=64, description="Pixel step for dimension alignment")
    min_dimension: int = Field(default=256, description="Minimum bucket dimension (pixels)")
    max_dimension: int = Field(default=1024, description="Maximum bucket dimension (pixels)")
    max_aspect_ratio: float = Field(default=2.0, description="Maximum aspect ratio (long/short)")

    @field_validator("step_size")
    @classmethod
    def validate_step_size(cls, v: int) -> int:
        if v < 8 or (v & (v - 1)) != 0:
            raise ValueError(
                f"step_size must be a power of 2 >= 8, got {v}. "
                f"Valid values: 8, 16, 32, 64, 128."
            )
        return v

    @field_validator("min_dimension")
    @classmethod
    def validate_min_dimension(cls, v: int) -> int:
        if v < 64:
            raise ValueError(f"min_dimension must be >= 64, got {v}.")
        return v

    @field_validator("max_dimension")
    @classmethod
    def validate_max_dimension(cls, v: int) -> int:
        if v > 4096:
            raise ValueError(f"max_dimension must be <= 4096, got {v}.")
        return v

    @field_validator("max_aspect_ratio")
    @classmethod
    def validate_max_aspect_ratio(cls, v: float) -> float:
        if v < 1.0:
            raise ValueError(f"max_aspect_ratio must be >= 1.0, got {v}.")
        return v


class TrainingHints(BaseModel):
    """Training hyperparameter suggestions for export config generation (Phase 8)."""
    model_config = ConfigDict(frozen=True)

    learning_rate_range: tuple[float, float] = Field(
        default=(1e-5, 1e-3),
        description="Suggested learning rate range (min, max)",
    )
    network_rank_range: tuple[int, int] = Field(
        default=(4, 128),
        description="Suggested LoRA network rank range (min, max)",
    )
    network_rank_default: int = Field(
        default=32,
        description="Default network rank suggestion",
    )
    network_alpha_ratio: float = Field(
        default=0.5,
        description="Suggested alpha = rank * ratio",
    )


class ModelProfile(BaseModel):
    """A target model profile defining resolution, bucket, and caption defaults.

    Profiles are frozen -- they cannot be modified after creation. Project
    overrides produce new profile instances via resolve_effective_config().
    """
    model_config = ConfigDict(frozen=True)

    name: str = Field(description="Machine-readable identifier (e.g., 'sd15', 'sdxl')")
    display_name: str = Field(description="Human-readable name")
    base_resolution: int = Field(description="Base resolution in pixels (defines pixel budget)")
    caption_style: Literal["booru", "natural_language"] = Field(
        description="Default caption format for this model type",
    )
    bucket_config: BucketConfig = Field(default_factory=BucketConfig)
    training_hints: TrainingHints = Field(default_factory=TrainingHints)
    description: str = Field(default="", description="Human-readable description")

    @model_validator(mode="after")
    def validate_resolution_alignment(self) -> ModelProfile:
        step = self.bucket_config.step_size
        if self.base_resolution % step != 0:
            raise ValueError(
                f"base_resolution ({self.base_resolution}) must be a multiple "
                f"of bucket_config.step_size ({step})."
            )
        return self

    @property
    def pixel_budget(self) -> int:
        return self.base_resolution * self.base_resolution


def generate_buckets(
    base_resolution: int,
    step_size: int = 64,
    min_dimension: int = 256,
    max_dimension: int | None = None,
    max_aspect_ratio: float = 2.0,
) -> list[tuple[int, int]]:
    """Generate valid bucket resolutions within a pixel budget.

    Follows the kohya/sd-scripts algorithm:
    1. Pixel budget = base_resolution^2
    2. For each width from min_dimension to max_dimension (step increments):
       a. Compute max valid height = floor(budget / width), snapped to step
       b. Clamp height to [min_dimension, max_dimension]
       c. Check aspect ratio constraint
       d. Add (width, height) and (height, width) if valid
    3. Return sorted, deduplicated bucket list

    Args:
        base_resolution: Defines pixel budget (base_resolution^2).
        step_size: Dimension alignment step (default 64).
        min_dimension: Smallest allowed dimension.
        max_dimension: Largest allowed dimension (default: 2 * base_resolution).
        max_aspect_ratio: Maximum ratio of long side to short side (default 2.0).

    Returns:
        Sorted list of (width, height) tuples.
    """
    if max_dimension is None:
        max_dimension = base_resolution * 2

    pixel_budget = base_resolution * base_resolution
    buckets: set[tuple[int, int]] = set()

    w = min_dimension
    while w <= max_dimension:
        # Maximum height that fits within pixel budget at this width
        h = (pixel_budget // w) // step_size * step_size
        h = min(h, max_dimension)

        if h >= min_dimension:
            ratio = max(w, h) / min(w, h)
            if ratio <= max_aspect_ratio:
                buckets.add((w, h))
                if w != h:
                    buckets.add((h, w))

        w += step_size

    return sorted(buckets)
```
