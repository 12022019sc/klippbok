```python
"""Per-project model config overrides and custom profile management.

The model config layer sits between built-in/custom profiles and the
project's effective configuration. It stores only the delta (overridden
fields), not a full profile copy. This means:
- Switching base profile applies new defaults to non-overridden fields
- Each field can be independently reset to the profile default
- The override file stays small and readable

Storage locations:
- Per-project overrides: .klippbok/model_config.json
- Custom profiles: ~/.klippbok/profiles/<name>.json
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, field_validator

from klippbok.config.model_defaults import BUILTIN_PROFILES
from klippbok.config.model_profiles import (
    BucketConfig,
    ModelProfile,
    TrainingHints,
)

KLIPPBOK_DIR = ".klippbok"
MODEL_CONFIG_FILE = "model_config.json"
USER_KLIPPBOK_DIR = Path.home() / ".klippbok"
USER_PROFILES_DIR = USER_KLIPPBOK_DIR / "profiles"


class ModelConfigOverride(BaseModel):
    """Per-project overrides on top of a model profile.

    Only overridden fields are stored. None means 'use profile default'.
    The profile_name field identifies which base profile these overrides
    apply to -- this prevents stale overrides when switching profiles.
    """
    model_config = ConfigDict(frozen=False)  # Mutable for field reset

    profile_name: str = Field(description="Base profile these overrides apply to")

    # Top-level overrides
    base_resolution: int | None = None
    caption_style: Literal["booru", "natural_language"] | None = None

    # Bucket config overrides (flattened for simpler JSON)
    bucket_step_size: int | None = None
    bucket_min_dimension: int | None = None
    bucket_max_dimension: int | None = None
    bucket_max_aspect_ratio: float | None = None

    # Training hint overrides
    learning_rate_range: tuple[float, float] | None = None
    network_rank_default: int | None = None
    network_alpha_ratio: float | None = None
```

The reconstruction validates the result (alignment check, etc.).

2. `load_model_config(project_dir: Path) -> ModelConfigOverride | None` -- Read .klippbok/model_config.json if it exists, return None otherwise.

4. `reset_override_field(project_dir: Path, field_name: str) -> ModelConfigOverride` -- Set a single override field to None (restoring profile default). Save and return updated overrides.
