```python
"""Image-to-bucket assignment by nearest aspect ratio.

Assigns each image to the closest valid training bucket, following the
NovelAI/kohya algorithm: argmin of absolute aspect ratio difference.
Buckets are generated by generate_buckets() from config/model_profiles.py.
"""

from __future__ import annotations


def assign_to_bucket(
    width: int,
    height: int,
    buckets: list[tuple[int, int]],
    max_aspect_ratio: float = 2.0,
) -> tuple[int, int] | None:
    """Find the nearest bucket for an image by aspect ratio.

    Args:
        width: Image width in pixels.
        height: Image height in pixels.
        buckets: Valid bucket dimensions from generate_buckets().
        max_aspect_ratio: Maximum allowed aspect ratio (long/short).
            Images exceeding this return None.

    Returns:
        (bucket_width, bucket_height) tuple, or None if the image's
        aspect ratio exceeds max_aspect_ratio or bucket list is empty.
    """
    if not buckets or height == 0:
        return None

    # Check if image has extreme aspect ratio
    image_max_ar = max(width, height) / min(width, height)
    if image_max_ar > max_aspect_ratio:
        return None

    image_ar = width / height
    return min(buckets, key=lambda b: abs(b[0] / b[1] - image_ar))


def needs_upscale(
    image_width: int,
    image_height: int,
    bucket_width: int,
    bucket_height: int,
) -> bool:
    """Check if image needs upscaling to fill assigned bucket.

    Returns True if either image dimension is smaller than the
    corresponding bucket dimension.

    Args:
        image_width: Source image width.
        image_height: Source image height.
        bucket_width: Assigned bucket width.
        bucket_height: Assigned bucket height.

    Returns:
        True if upscaling would be needed.
    """
    return image_width < bucket_width or image_height < bucket_height
```

```python
"""Image quality assessment -- blur detection via Laplacian variance.

Uses numpy + scipy for Laplacian convolution. No OpenCV dependency.
The Laplacian variance measures edge sharpness: higher values mean
sharper images, lower values mean more blur.

Threshold 100.0 is a fixed pass/fail value per project decisions.
Quality checks are advisory only -- they never block import.
"""

from __future__ import annotations

import numpy as np
from PIL import Image
from scipy.signal import convolve2d

# Standard 3x3 Laplacian kernel (same operator as cv2.Laplacian)
_LAPLACIAN_KERNEL = np.array([
    [0,  1,  0],
    [1, -4,  1],
    [0,  1,  0],
], dtype=np.float64)

BLUR_THRESHOLD: float = 100.0
"""Fixed pass/fail threshold for blur detection. Images with Laplacian
variance below this value are flagged as blurry. Not user-adjustable."""


def compute_blur_score(img: Image.Image) -> float:
    """Compute Laplacian variance as a blur/sharpness metric.

    Higher values indicate sharper images. Lower values indicate blur.
    Uses grayscale conversion internally.

    Args:
        img: PIL Image in any mode (RGB, RGBA, L, etc.).

    Returns:
        Laplacian variance as a float. Higher = sharper.
    """
    gray = np.array(img.convert("L"), dtype=np.float64)
    laplacian = convolve2d(gray, _LAPLACIAN_KERNEL, mode="valid")
    return float(laplacian.var())


def is_blurry(img: Image.Image) -> bool:
    """Check if an image is blurry (Laplacian variance below threshold).

    Args:
        img: PIL Image in any mode.

    Returns:
        True if the image's blur score is below BLUR_THRESHOLD.
    """
    return compute_blur_score(img) < BLUR_THRESHOLD
```
